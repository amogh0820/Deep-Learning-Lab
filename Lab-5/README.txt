// Class representing a complex number with real and imaginary parts
class ComplexNumber {
    private double real;
    private double imaginary;

    // Constructor to initialize real and imaginary parts
    public ComplexNumber(double real, double imaginary) {
        this.real = real;
        this.imaginary = imaginary;
    }

    // Method to add another ComplexNumber and return a newly constructed ComplexNumber object
    public ComplexNumber add(ComplexNumber c) {
        double newReal = this.real + c.real;
        double newImaginary = this.imaginary + c.imaginary;

        // Construct and return a new object representing the sum
        return new ComplexNumber(newReal, newImaginary);
    }

    // Helper method to display the complex number format (a + bi)
    public void display() {
        if (imaginary >= 0) {
            System.out.println(real + " + " + imaginary + "i");
        } else {
            System.out.println(real + " - " + Math.abs(imaginary) + "i");
        }
    }
}

public class Main {
    public static void main(String[] args) {
        // Create two initial ComplexNumber objects
        ComplexNumber c1 = new ComplexNumber(4.5, 2.5);
        ComplexNumber c2 = new ComplexNumber(1.5, 3.5);

        System.out.print("First Complex Number: ");
        c1.display();

        System.out.print("Second Complex Number: ");
        c2.display();

        // Call the add method which returns a new ComplexNumber object
        ComplexNumber result = c1.add(c2);

        System.out.print("Sum (New Object): ");
        result.display();
    }
}
